# Git commands to push test-pipeline branch to main

# 1. Check current branch and status
git status
git branch

# 2. Switch to main branch
git checkout main

# 3. Pull latest changes from remote main
git pull origin main

# 4. Merge test-pipeline branch into main
git merge test-pipeline

# 5. Push changes to remote main branch
git push origin main
